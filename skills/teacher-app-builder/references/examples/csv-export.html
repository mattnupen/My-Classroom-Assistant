let csvText = '';  // in memory only — never saved
document.getElementById('btn-build').addEventListener('click', () => {
  const lines = document.getElementById('input').value.split('\n').filter(Boolean);
  // ADAPT: turn `lines` into this app's rows
  const rows = [['Item', 'Count']].concat(lines.map((l, i) => [l, i + 1]));
  csvText = rows.map(r => r.map(c => `"${String(c).replace(/"/g, '""')}"`).join(',')).join('\n');
  const preview = document.getElementById('preview');
  preview.textContent = csvText;  // textContent, never innerHTML, for anything the teacher typed
  preview.hidden = false;
  document.getElementById('btn-copy').disabled = false;
  document.getElementById('btn-download').disabled = false;
});
document.getElementById('btn-copy').addEventListener('click', async (e) => {
  try { await navigator.clipboard.writeText(csvText); e.target.textContent = 'Copied'; }
  catch { e.target.textContent = 'Select the text below and copy it'; }
});
document.getElementById('btn-download').addEventListener('click', () => {
  const a = document.createElement('a');
  a.href = URL.createObjectURL(new Blob([csvText], { type: 'text/csv' }));
  a.download = 'output.csv';  // ADAPT: a plain file name, e.g. 'exit-ticket-tally.csv'
  a.click();
  URL.revokeObjectURL(a.href);
});
